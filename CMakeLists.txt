cmake_minimum_required(VERSION 3.20)
project(beat_echo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BE_SANITIZERS "Enable address/undefined behavior sanitizers" OFF)
add_library(beat_echo_core core/src/engine.cpp core/src/synth.cpp)
target_include_directories(beat_echo_core PUBLIC core/include)
target_compile_options(beat_echo_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(BE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(beat_echo_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(beat_echo_core PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(beat_echo_tests tests/test_core.cpp)
target_link_libraries(beat_echo_tests PRIVATE beat_echo_core)
target_compile_options(beat_echo_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME core COMMAND beat_echo_tests)
add_executable(render_demo tests/render_demo.cpp)
target_link_libraries(render_demo PRIVATE beat_echo_core)
